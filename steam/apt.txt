# Native Steam (hosts with nativeSteam = "yes" in facts.nix; ws help
# plan-legion-go): the apt Steam instead of the Flathub flatpak, which the
# Game Mode session cannot use (it calls host helpers such as
# steamos-session-select). bootstrap.sh installs it, ws check apt compares.
# The installer asks to accept the Steam license during bootstrap.sh.

# 32-bit libraries of the Steam client and of games
arch:i386
# With the recommends of steam-libs and steam-libs-i386 (rec:, as Ubuntu
# installs Steam): the 32-bit SDL2, ALSA plugins (sound of 32-bit games),
# VA-API/VDPAU, GTK3 and codecs.
rec:steam-installer
# Of those recommends, the ones Proton cannot do without, listed so that
# ws check apt sees them: 32-bit Vulkan for DXVK/VKD3D, EGL and GBM of the
# 32-bit client (steam-libs-i386 itself pulls only GL).
mesa-vulkan-drivers
mesa-vulkan-drivers:i386
libegl1:i386
libgbm1:i386
